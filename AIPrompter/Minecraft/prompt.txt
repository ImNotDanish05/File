### ⚠️ CRITICAL SECTION — MANDATORY HELPER FUNCTIONS (READ EVERY WORD):

The following helper functions contain ALL the logic for placing blocks safely. You MUST copy them **exactly as written** into your script. Do NOT modify them, do NOT rewrite them, do NOT inline their logic.

```python
def place_block(x, y, z, block_id, properties=None):
    """THE ONLY permitted way to place a block. Never call BlockState directly."""
    if not (0 <= int(x) < width and 0 <= int(y) < height and 0 <= int(z) < length):
        return
    safe_props = {}
    if properties:
        for k, v in properties.items():
            if isinstance(v, bool):
                safe_props[str(k)] = "true" if v else "false"
            else:
                safe_props[str(k)] = str(v).lower()
    try:
        if safe_props:
            reg[int(x), int(y), int(z)] = BlockState(block_id, properties=safe_props)
        else:
            reg[int(x), int(y), int(z)] = BlockState(block_id)
    except Exception:
        try:
            reg[int(x), int(y), int(z)] = BlockState(block_id)
        except Exception:
            pass

def fill_box(x1, y1, z1, x2, y2, z2, block_id, properties=None):
    """Fill a rectangular region. Always use this instead of nested loops with BlockState."""
    for x in range(min(x1, x2), max(x1, x2) + 1):
        for y in range(min(y1, y2), max(y1, y2) + 1):
            for z in range(min(z1, z2), max(z1, z2) + 1):
                place_block(x, y, z, block_id, properties)

def fill_hollow(x1, y1, z1, x2, y2, z2, block_id, properties=None):
    """Fill only the outer shell of a rectangular region."""
    for x in range(min(x1, x2), max(x1, x2) + 1):
        for y in range(min(y1, y2), max(y1, y2) + 1):
            for z in range(min(z1, z2), max(z1, z2) + 1):
                if (x in (x1, x2) or y in (y1, y2) or z in (z1, z2)):
                    place_block(x, y, z, block_id, properties)

def place_weighted(x, y, z, palette):
    """
    Place a block chosen from a weighted palette.
    palette is a list of (block_id, weight) tuples. Example:
      palette = [("minecraft:stone", 5), ("minecraft:cobblestone", 3), ("minecraft:andesite", 2)]
    """
    blocks = [b for b, _ in palette]
    weights = [w for _, w in palette]
    chosen = random.choices(blocks, weights=weights, k=1)[0]
    place_block(x, y, z, chosen)

def fill_box_weighted(x1, y1, z1, x2, y2, z2, palette):
    """Fill a rectangular region using a weighted palette."""
    for x in range(min(x1, x2), max(x1, x2) + 1):
        for y in range(min(y1, y2), max(y1, y2) + 1):
            for z in range(min(z1, z2), max(z1, z2) + 1):
                place_weighted(x, y, z, palette)
```
